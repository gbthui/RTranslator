cmake_minimum_required(VERSION 3.22)
project(rtranslator_qwen LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_C_VISIBILITY_PRESET hidden)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)

if(QWEN_RUNTIME STREQUAL "onnx")
    set(SHERPA_ONNX_ENABLE_JNI OFF CACHE BOOL "" FORCE)
    set(SHERPA_ONNX_ENABLE_C_API OFF CACHE BOOL "" FORCE)
    set(SHERPA_ONNX_ENABLE_BINARY OFF CACHE BOOL "" FORCE)
    set(SHERPA_ONNX_ENABLE_PORTAUDIO OFF CACHE BOOL "" FORCE)
    set(SHERPA_ONNX_ENABLE_WEBSOCKET OFF CACHE BOOL "" FORCE)
    set(SHERPA_ONNX_ENABLE_TTS OFF CACHE BOOL "" FORCE)
    set(SHERPA_ONNX_ENABLE_SPEAKER_DIARIZATION OFF CACHE BOOL "" FORCE)
    set(SHERPA_ONNX_USE_PRE_INSTALLED_ONNXRUNTIME_IF_AVAILABLE ON CACHE BOOL "" FORCE)
    add_subdirectory("${RUNTIME_SOURCE}" sherpa)
    set(runtime rtranslator_qwen_onnx)
    add_library(${runtime} SHARED sherpa_jni.cpp)
    target_include_directories(${runtime} PRIVATE "${RUNTIME_SOURCE}" "$ENV{SHERPA_ONNXRUNTIME_INCLUDE_DIR}")
    target_link_libraries(${runtime} PRIVATE sherpa-onnx-core)
else()
    set(TRANSCRIBE_BUILD_SHARED OFF CACHE BOOL "" FORCE)
    set(TRANSCRIBE_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(TRANSCRIBE_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(TRANSCRIBE_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
    set(TRANSCRIBE_INSTALL OFF CACHE BOOL "" FORCE)
    set(TRANSCRIBE_USE_OPENMP OFF CACHE BOOL "" FORCE)
    set(TRANSCRIBE_USE_SYSTEM_BLAS OFF CACHE BOOL "" FORCE)
    set(TRANSCRIBE_METAL OFF CACHE BOOL "" FORCE)
    set(TRANSCRIBE_CUDA OFF CACHE BOOL "" FORCE)
    set(TRANSCRIBE_VULKAN ${QWEN_GGUF_VULKAN} CACHE BOOL "" FORCE)
    set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
    set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
    set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
    set(TRANSCRIBE_X86_CONSERVATIVE ON CACHE BOOL "" FORCE)
    if(ANDROID_ABI STREQUAL "arm64-v8a")
        set(GGML_CPU_ARM_ARCH "armv8-a" CACHE STRING "" FORCE)
    endif()
    add_subdirectory("${RUNTIME_SOURCE}" transcribe)
    target_include_directories(transcribe PUBLIC "${RUNTIME_SOURCE}/include")
    if(QWEN_GGUF_VULKAN)
        set(runtime rtranslator_qwen_gguf_vulkan)
    else()
        set(runtime rtranslator_qwen_gguf_cpu)
    endif()
    add_library(${runtime} SHARED gguf_jni.cpp)
    target_compile_definitions(${runtime} PRIVATE QWEN_GGUF_VULKAN=$<BOOL:${QWEN_GGUF_VULKAN}>)
    target_include_directories(${runtime} PRIVATE "${RUNTIME_SOURCE}/ggml/include")
    target_link_libraries(${runtime} PRIVATE transcribe ggml)
endif()

if(ANDROID)
    target_link_libraries(${runtime} PRIVATE log)
else()
    find_package(JNI REQUIRED)
    target_include_directories(${runtime} PRIVATE ${JNI_INCLUDE_DIRS})
endif()
target_link_options(${runtime} PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/exports.map"
    "LINKER:--exclude-libs,ALL" "LINKER:-Bsymbolic" "LINKER:-z,max-page-size=16384")
set_target_properties(${runtime} PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/out")
